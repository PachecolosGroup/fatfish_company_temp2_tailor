        $$(".idioma button").forEach(function (b) {
          b.addEventListener("click", function () {
            idioma = b.dataset.lang;
            try {
              localStorage.setItem("lov-idioma", idioma);
            } catch (e) {}
            aplicarIdioma();
          });
        });

        /* --- Lightbox --- */
        var lb = $("#lb"),
          lbImg = $("#lb-img"),
          lbPie = $("#lb-pie"),
          actual = 0,
          previo = null;

        function abrirLB(i) {
          actual = (i + FOTOS.length) % FOTOS.length;
          var f = FOTOS[actual],
            alt = f[idioma === "es" ? 1 : 2];
          lbImg.src = "img/" + f[0] + ".webp";
          lbImg.alt = alt;
          lbPie.textContent =
            alt + "  ·  " + (actual + 1) + " / " + FOTOS.length;
          lb.classList.add("abierto");
          document.body.style.overflow = "hidden";
          $("#lb-cerrar").focus();
        }
        function cerrarLB() {
          lb.classList.remove("abierto");
          document.body.style.overflow = "";
          if (previo) previo.focus();
        }
        grid.addEventListener("click", function (e) {
          var b = e.target.closest(".foto");
          if (!b) return;
          previo = b;
          abrirLB(parseInt(b.dataset.i, 10));
        });
        $("#lb-cerrar").addEventListener("click", cerrarLB);
        $("#lb-prev").addEventListener("click", function () {
          abrirLB(actual - 1);
        });
        $("#lb-sig").addEventListener("click", function () {
          abrirLB(actual + 1);
        });
        lb.addEventListener("click", function (e) {
          if (e.target === lb) cerrarLB();
        });
        document.addEventListener("keydown", function (e) {
          if (!lb.classList.contains("abierto")) return;
          if (e.key === "Escape") cerrarLB();
          else if (e.key === "ArrowLeft") abrirLB(actual - 1);
          else if (e.key === "ArrowRight") abrirLB(actual + 1);
        });

        /* --- Calendario --- */
        var tipoCita = "video";

        function calendarioListo() {
          return (
            !!window.Calendly && CALENDLY[tipoCita].indexOf("TU-USUARIO") === -1
          );
        }
        function montarCalendario() {
          var widget = $("#cal-widget"),
            espera = $("#cal-espera");
          if (!calendarioListo()) {
            widget.hidden = true;
            espera.hidden = false;
            return;
          }
          widget.hidden = false;
          espera.hidden = true;
          widget.innerHTML = "";
          window.Calendly.initInlineWidget({
            url:
              CALENDLY[tipoCita] +
              "?hide_gdpr_banner=1&background_color=0B0B0D&text_color=F4F0E8&primary_color=C9A227",
            parentElement: widget,
          });
        }
        $$(".cal-tipo button").forEach(function (b) {
          b.addEventListener("click", function () {
            tipoCita = b.dataset.tipo;
            $$(".cal-tipo button").forEach(function (x) {
              x.setAttribute(
                "aria-pressed",
                String(x.dataset.tipo === tipoCita),
              );
            });
            montarCalendario();
          });
        });
        // Cuadrícula decorativa del panel de espera
        (function () {
          var r = $(".cal-espera .rejilla");
          for (var i = 0; i < 35; i++) {
            var c = document.createElement("i");
            if ([9, 10, 16, 17, 18, 23, 25, 31].indexOf(i) !== -1)
              c.className = "on";
            r.appendChild(c);
          }
        })();
        window.addEventListener("load", function () {
          setTimeout(montarCalendario, 400);
        });

        /* --- Aparición al hacer scroll --- */
        var revs = $$(".rev");
        if (reduce || !("IntersectionObserver" in window)) {
          revs.forEach(function (el) {
            el.classList.add("dentro");
          });
        } else {
          var obs = new IntersectionObserver(
            function (es) {
              es.forEach(function (e) {
                if (e.isIntersecting) {
                  e.target.classList.add("dentro");
                  obs.unobserve(e.target);
                }
              });
            },
            { threshold: 0.12, rootMargin: "0px 0px -50px 0px" },
          );
          revs.forEach(function (el) {
            obs.observe(el);
          });
        }

        $("#anio").textContent = new Date().getFullYear();
        aplicarIdioma();
        montarCalendario();
      })();
    </script>
